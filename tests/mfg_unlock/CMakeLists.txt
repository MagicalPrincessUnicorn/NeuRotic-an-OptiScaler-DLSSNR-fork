cmake_minimum_required(VERSION 3.24)
project(neurotic_experimental_mfg LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(nr_experimental_mfg_tests ../nr_experimental_mfg.cpp)
add_test(NAME nr_experimental_mfg_tests COMMAND nr_experimental_mfg_tests)
add_executable(nr_experimental_mfg_host_tests ../nr_experimental_mfg_host.cpp)
add_test(NAME nr_experimental_mfg_host_tests COMMAND nr_experimental_mfg_host_tests)
set(upstream ${CMAKE_CURRENT_SOURCE_DIR}/../../external/mfg_ampere/contrib/validation/tests)
add_executable(nr_experimental_mfg_transactions ../nr_experimental_mfg_transactions.cpp)
target_compile_definitions(nr_experimental_mfg_transactions PRIVATE NR_EXPERIMENTAL_MFG_TEST)
target_include_directories(nr_experimental_mfg_transactions PRIVATE ${upstream}/provider_mocks)
add_test(NAME nr_experimental_mfg_transactions COMMAND nr_experimental_mfg_transactions)
foreach(name architecture capability_policy ptx_retarget provider_profile provider_state blackwell_temporal endpoint_backend)
    add_executable(${name}_test ${upstream}/${name}_test.cpp)
    target_compile_definitions(${name}_test PRIVATE NR_EXPERIMENTAL_MFG_TEST)
    target_include_directories(${name}_test PRIVATE ${upstream}/provider_mocks)
    add_test(NAME ${name}_test COMMAND ${name}_test)
endforeach()
if(MSVC)
    # Keep assert-based policy tests active in Release too.
    target_compile_options(nr_experimental_mfg_tests PRIVATE /UNDEBUG)
    target_compile_options(nr_experimental_mfg_host_tests PRIVATE /UNDEBUG)
endif()
